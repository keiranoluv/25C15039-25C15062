\begin{algorithmic}[1]
\REQUIRE Training triples $\mathcal{T}_{\mathrm{train}}$;
    modalities $\mathcal{M}=\{S,V,D\}$; $\tau,\alpha,\beta,\gamma$.
\ENSURE A trained structure-only student $U_\psi$.
\STATE \textbf{Multimodal teacher pre-training}
\STATE Pre-train each teacher $T_m$, $m\in\mathcal{M}$,
    by minimizing entity-prediction cross-entropy.
\STATE Freeze the teachers; initialize student $U_\psi$
    and policy $\pi_\theta$.
\STATE \textbf{Student and policy training}
\FOR{each training epoch}
    \FOR{each batch $\mathcal{B}\subset\mathcal{T}_{\mathrm{train}}$}
        \FOR{each triple $(h_i,r_i,t_i)\in\mathcal{B}$}
            \STATE Obtain teacher logits $F_i^S,F_i^V,F_i^D$
                and student logits $F_i^{\mathrm{stu}}$.
            \STATE Construct $s_i=[F_i^S\Vert F_i^V\Vert F_i^D]$;
                sample $a_i\sim\pi_\theta(\cdot\mid s_i)$.
            \STATE Let $M_i$ be the nonempty teacher subset
                selected by $a_i$ (one of seven combinations).
            \STATE $F_i^{\mathrm{tea}}=|M_i|^{-1}
                \sum_{m\in M_i}F_i^m$;
                $F_i^{\mathrm{avg}}=\frac{1}{3}
                \sum_{m\in\mathcal{M}}F_i^m$.
            \STATE Form $N_i=\{e:(h_i,r_i,e)
                \in\mathcal{T}_{\mathrm{train}}\}$.
            \STATE Apply temperature-$\tau$ softmax to
                $F_i^{\mathrm{tea}}$ and $F_i^{\mathrm{stu}}$.
            \STATE Compute $\mathcal{L}_{\mathrm{NEKD},i}$
                from mean neighbor probabilities and
                $\mathcal{L}_{\mathrm{NNKD},i}$ from normalized
                non-neighbor distributions.
            \STATE $\mathcal{L}_{\mathrm{NDKD},i}
                =\alpha\mathcal{L}_{\mathrm{NEKD},i}
                +\beta\mathcal{L}_{\mathrm{NNKD},i}$.
            \STATE Set $R_i=1$ if
                $\mathrm{CE}(F_i^{\mathrm{tea}},t_i)
                <\mathrm{CE}(F_i^{\mathrm{stu}},t_i)$;
                otherwise set $R_i=-10$.
            \STATE Compute baseline reward $\widehat{R}_i$
                using the same rule with $F_i^{\mathrm{avg}}$.
            \STATE $\delta_i=\operatorname{stopgrad}
                (R_i-\widehat{R}_i)$.
        \ENDFOR
        \STATE $\mathcal{L}_{\mathrm{upd}}
            =\frac{1}{|\mathcal{B}|}\sum_{i\in\mathcal{B}}
            [\mathrm{CE}(F_i^{\mathrm{stu}},t_i)
            +\gamma\mathcal{L}_{\mathrm{NDKD},i}]$.
        \STATE $\mathcal{L}_{\mathrm{RC}}
            =-\frac{1}{|\mathcal{B}|}\sum_{i\in\mathcal{B}}
            \delta_i\log\pi_\theta(a_i\mid s_i)$.
        \STATE Update $\psi$ using $\mathcal{L}_{\mathrm{upd}}$
            and $\theta$ using $\mathcal{L}_{\mathrm{RC}}$;
            keep teacher parameters fixed.
    \ENDFOR
\ENDFOR
\STATE \textbf{Inference with the student only}
\STATE For each query $(h,r,?)$, score all candidate entities
    with $U_\psi$ and rank them in descending score order.
\STATE Handle head prediction using reciprocal queries;
    teachers and policy are not used at inference.
\end{algorithmic}
